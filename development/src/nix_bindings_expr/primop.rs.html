12/// A primop error that is not memoized in the thunk that triggered it,
13/// allowing the thunk to be forced again.
14///
15/// Since [Nix 2.34](https://nix.dev/manual/nix/2.34/release-notes/rl-2.34.html#c-api-changes),
16/// primop errors are memoized by default: once a thunk fails, forcing it
17/// again returns the same error. Use `RecoverableError` for errors that
18/// are transient, so the caller can retry.
19///
20/// On Nix < 2.34, all errors are already recoverable, so this type has
21/// no additional effect.
22///
23/// Available since nix-bindings-expr 0.2.1.
24#[derive(Debug)]
25pub struct RecoverableError(String);
26
27impl RecoverableError {
28    pub fn new(msg: impl Into<String>) -> Self {
29        RecoverableError(msg.into())
30    }
31}
32
33impl std::fmt::Display for RecoverableError {
34    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
35        self.0.fmt(f)
36    }
37}
38
39impl std::error::Error for RecoverableError {}

41/// Metadata for a primop, used with `PrimOp::new`.
42pub struct PrimOpMeta<'a, const N: usize> {
43    /// Name of the primop. Note that primops do not have to be registered as
44    /// builtins. Nonetheless, a name is required for documentation purposes, e.g.
45    /// :doc in the repl.
46    pub name: &'a CStr,
47
48    /// Documentation for the primop. This is displayed in the repl when using
49    /// :doc. The format is markdown.
50    pub doc: &'a CStr,
51
52    /// The number of arguments the function takes, as well as names for the
53    /// arguments, to be presented in the documentation (if applicable, e.g.
54    /// :doc in the repl).
55    pub args: [&'a CStr; N],
56}

58pub struct PrimOp {
59    ptr: *mut raw::PrimOp,
60}
61impl Drop for PrimOp {
62    fn drop(&mut self) {
63        unsafe {
64            raw::gc_decref(null_mut(), self.ptr as *mut c_void);
65        }
66    }
67}
68impl PrimOp {
69    /// Create a new primop with the given metadata and implementation.
70    ///
71    /// When `f` returns an `Err`, the error is propagated to the Nix evaluator.
72    /// To return a [recoverable error](RecoverableError), include it in the
73    /// error chain (e.g. `Err(RecoverableError::new("...").into())`).
74    pub fn new<const N: usize>(
75        meta: PrimOpMeta<N>,
76        f: Box<dyn Fn(&mut EvalState, &[Value; N]) -> Result<Value>>,
77    ) -> Result<PrimOp> {
78        assert!(N != 0);
79
80        let mut args = Vec::new();
81        for arg in meta.args {
82            args.push(arg.as_ptr());
83        }
84        args.push(null());
85
86        // Primops weren't meant to be dynamically created, as of writing.
87        // This leaks, and so do the primop fields in Nix internally.
88        let user_data = {
89            // We'll be leaking this Box.
90            // TODO: Use the GC with finalizer, if possible.
91            let user_data = ManuallyDrop::new(Box::new(PrimOpContext {
92                arity: N,
93                function: Box::new(move |eval_state, args| f(eval_state, args.try_into().unwrap())),
94            }));
95            user_data.as_ref() as *const PrimOpContext as *mut c_void
96        };
97        let mut ctx = Context::new();
98        let op = unsafe {
99            check_call!(raw::alloc_primop(
100                &mut ctx,
101                FUNCTION_ADAPTER,
102                N as c_int,
103                meta.name.as_ptr(),
104                args.as_mut_ptr(), /* TODO add an extra const to bindings to avoid mut here. */
105                meta.doc.as_ptr(),
106                user_data
107            ))?
108        };
109        Ok(PrimOp { ptr: op })
110    }
111
112    /// Creates a [function][`crate::value::ValueType::Function`] [`Value`]
113    /// backed by this primop, in the given [`EvalState`].
114    ///
115    /// Consumes the [`PrimOp`]; the resulting [`Value`] keeps the primop alive
116    /// via Nix's GC.
117    #[doc(alias = "make_primop")]
118    #[doc(alias = "create_function")]
119    #[doc(alias = "builtin")]
120    #[doc(alias = "init_primop")]
121    pub fn to_value(self, eval_state: &mut EvalState) -> Result<Value> {
122        let value = eval_state.new_value_uninitialized()?;
123        unsafe {
124            check_call!(raw::init_primop(
125                &mut eval_state.context,
126                value.raw_ptr(),
127                self.ptr
128            ))?;
129        }
130        Ok(value)
131    }
132
133    /// Registers this primop as a global builtin.
134    ///
135    /// This only affects [`EvalState`]s created *after* the call.
136    ///
137    /// The same primop must not also be attached to a value with [`PrimOp::to_value`].
138    // Implementation note:
139    // `Drop` still
140    // calls `nix_gc_decref` to release our reference — the primop content
141    // has been moved into Nix's global registry by then, per the C API
142    // contract.
143    #[doc(alias = "nix_register_primop")]
144    pub fn register(self) -> Result<()> {
145        let mut ctx = Context::new();
146        unsafe {
147            check_call!(raw::register_primop(&mut ctx, self.ptr))?;
148        }
149        Ok(())
150    }
151}

153/// The user_data for our Nix primops
154struct PrimOpContext {
155    arity: usize,
156    function: Box<dyn Fn(&mut EvalState, &[Value]) -> Result<Value>>,
157}
158
159unsafe extern "C" fn function_adapter(
160    user_data: *mut ::std::os::raw::c_void,
161    context_out: *mut raw_util::c_context,
162    state: *mut raw::EvalState,
163    args: *mut *mut raw::Value,
164    ret: *mut raw::Value,
165) {
166    let primop_info = (user_data as *const PrimOpContext).as_ref().unwrap();
167    let mut eval_state = EvalState::from_raw_borrowed(
168        NonNull::new(state).expect("Nix primop callback given null EvalState pointer"),
169    );
170    let args_raw_slice = unsafe { std::slice::from_raw_parts(args, primop_info.arity) };
171    let args_vec: Vec<Value> = args_raw_slice
172        .iter()
173        .map(|v| Value::new_borrowed(*v))
174        .collect();
175    let args_slice = args_vec.as_slice();
176
177    let r = primop_info.function.as_ref()(&mut eval_state, args_slice);
178
179    // Do not keep. See `EvalState::from_raw_borrowed` safety section.
180    // The original Nix C API `EvalState *` wrapper may be transient.
181    drop(eval_state);
182
183    match r {
184        Ok(v) => unsafe {
185            raw::copy_value(context_out, ret, v.raw_ptr());
186        },
187        Err(e) => unsafe {
188            let err_code = error_code(&e);
189            let cstr = CString::new(e.to_string()).unwrap_or_else(|_e| {
190                CString::new("<rust nix-expr application error message contained null byte>")
191                    .unwrap()
192            });
193            raw_util::set_err_msg(context_out, err_code, cstr.as_ptr());
194        },
195    }
196}
197
198fn error_code(e: &anyhow::Error) -> raw_util::err {
199    #[cfg(nix_at_least = "2.34.0pre")]
200    if e.downcast_ref::<RecoverableError>().is_some() {
201        return raw_util::err_NIX_ERR_RECOVERABLE;
202    }
203    raw_util::err_NIX_ERR_UNKNOWN
204}

206static FUNCTION_ADAPTER: raw::PrimOpFun = Some(function_adapter);
